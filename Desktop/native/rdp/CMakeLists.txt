cmake_minimum_required(VERSION 3.25)
project(serverdash_rdp_display VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This display adapter requires Windows; the C ABI is platform neutral.")
endif()
find_package(FreeRDP 3.31.0 EXACT CONFIG REQUIRED)
add_library(serverdash_rdp_display SHARED display.cpp)
target_compile_features(serverdash_rdp_display PRIVATE cxx_std_17)
target_compile_definitions(serverdash_rdp_display PRIVATE SD_RDP_BUILD WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(serverdash_rdp_display PUBLIC include)
target_link_libraries(serverdash_rdp_display PRIVATE freerdp d3d11 dxgi user32)
add_executable(serverdash_rdp_smoke smoke.cpp)
target_link_libraries(serverdash_rdp_smoke PRIVATE serverdash_rdp_display user32)
enable_testing()
add_test(NAME native_display COMMAND serverdash_rdp_smoke)
install(TARGETS serverdash_rdp_display serverdash_rdp_smoke RUNTIME DESTINATION bin)
install(FILES include/serverdash_rdp.h DESTINATION include)
